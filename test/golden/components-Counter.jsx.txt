import { useState, useEffect, useCallback } from 'react';
import confetti from 'canvas-confetti';

export default function Counter({ label = "Count", step = 1 }) {
  const [n, setN] = useState(0);
  useEffect(() => {
    document.title = `${label}: ${n}`;
  }, [label, n]);
  const bump = useCallback((by) => { setN(n + (by)); confetti() }, [n]);
  return (
    <div className="counter">
      <h2>{label}: {n}</h2>
      <button className="btn primary" onClick={() => bump(step)}>+</button>
      <button className="btn" onClick={() => setN(n - (step))}>-</button>
      {n>0 ? (
        <p className="hint">keep going</p>
      ) : (
        <p className="hint">click +</p>
      )}
    </div>
  );
}
