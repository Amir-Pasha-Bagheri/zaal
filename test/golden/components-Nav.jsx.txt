import logo from '../img/logo.png';
import { useTheme } from '../stores/Theme.jsx';
import { useCaught } from '../stores/Caught.jsx';
import { Link } from '../router.jsx';

export default function Nav() {
  const { mode, toggle } = useTheme();
  const { total } = useCaught();
  return (
    <nav className="nav">
      <img className="mark" src={logo} alt="Zaal" width={22} height={22} />
      <Link to="/">Home</Link>
      <Link to="/dex">Pokédex</Link>
      <Link to="/stats">Stats</Link>
      <Link to="/about">About</Link>
      <span className="spacer" />
      {total ? (
        <span className="pill">{total} caught</span>
      ) : null}
      <button className="btn icon" onClick={toggle} title="theme">{mode == "dark" ? "☀" : "☾"}</button>
    </nav>
  );
}
