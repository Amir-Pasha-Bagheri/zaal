import { useState, useEffect, useMemo, useCallback } from 'react';
import Nav from '../components/Nav.jsx';
import Field from '../components/Field.jsx';
import Spinner from '../components/Spinner.jsx';
import Badge from '../components/Badge.jsx';
import { FALLBACK, API, art } from '../data/dex.jsx';
import { useCaught } from '../stores/Caught.jsx';
import { Link } from '../router.jsx';

export default function Dex() {
  const [q, setQ] = useState("");
  const [list, setList] = useState([]);
  const [status, setStatus] = useState("loading");
  const { has, toggle } = useCaught();
  const load = useCallback((signal) => { setStatus("loading"); fetch(`${API}?limit=24`, { signal }).then(r => r.json()).then(d => { setList(d.results); setStatus("live") }).catch(e => { if (e.name === "AbortError") return; setList(FALLBACK); setStatus("offline") }) }, []);
  useEffect(() => {
    const ctrl = new AbortController();
    load(ctrl.signal);
    return () => ctrl.abort();
  }, []);
  const shown = useMemo(() => list.filter(p => p.name.includes(q.trim().toLowerCase())), [q, list]);
  return (
    <main className="page">
      <Nav />
      <h1>Pokédex</h1>
      <form className="row search">
        <Field label="search" value={q} onInput={(e) => setQ(e.target.value)} placeholder="pika…" />
      </form>
      {status == "loading" ? (
        <Spinner label="asking the API…" />
      ) : !shown.length ? (
        <p className="hint">nothing matches “{q}”</p>
      ) : (
        <div className="grid">
          {shown.map((p, i) => (
            <div key={p.name} className="tile">
              <Link to={`/dex/${p.name}`}>
                <img className="sprite" src={art(p.name)} alt={p.name} loading="lazy" />
                <span>{p.name}</span>
              </Link>
              <button className="btn icon" onClick={() => toggle(p.name)} title="catch">{has(p.name) ? "★" : "☆"}</button>
            </div>
          ))}
        </div>
      )}
      {status == "offline" ? (
        <Badge text="offline — showing bundled data" />
      ) : null}
    </main>
  );
}
