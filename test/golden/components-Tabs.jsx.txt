import { useState, useMemo, useContext, createContext } from 'react';

export const TabsContext = createContext(null);

export function useTabs() {
  const value = useContext(TabsContext);
  if (!value) throw new Error('useTabs() used outside a `provide Tabs`');
  return value;
}

export default function Tabs({ labels, children }) {
  const [open, setOpen] = useState(0);
  const tabs = useMemo(() => ({ open, setOpen }), [open]);
  return (
    <TabsContext.Provider value={tabs}>
      <div>
        <div className="tablist">
          {labels.map((label, i) => (
            <button key={i} className="tab" onClick={() => setOpen(i)} aria-selected={open == i}>{label}</button>
          ))}
        </div>
        {children}
      </div>
    </TabsContext.Provider>
  );
}
export function Tab({ at, children }) {
  const { open } = useTabs();
  return (
    <>
      {open == at ? (
        <div className="tabpanel">
          {children}
        </div>
      ) : null}
    </>
  );
}
